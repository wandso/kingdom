local cin,cout,th_i = ...
require "love.filesystem"
require "love.audio"
require "love.sound"
local file_count = 0
while true do
    -- get params
    local file = cin:demand()
    if file == 'QUIT' then goto quit end
    local mode = cin:demand()
    local id = cin:demand()
    
    if not love.filesystem.isFile(file) then
        cout:push({'ERROR','Not a file',file})
    else
        local ok, result = pcall(love.audio.newSource, file, mode)
        collectgarbage()
        if ok and result then
            cout:push({'OK',result,id})
            file_count = file_count + 1
        else
            cout:push({'ERROR',result,file})
        end
    end
end
::quit::
cout:supply({'DONE'})
